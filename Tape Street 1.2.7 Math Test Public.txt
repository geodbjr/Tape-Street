        self.entropy_pool = low_level.hash_secret_raw(
            secret=password.encode(),
            salt=salt,
            time_cost=3,
            memory_cost=65536,  # 64MB RAM
            parallelism=4,
            hash_len=64,
            type=low_level.Type.ID
        )
        
        self.cipher_key = self.entropy_pool[:32]
        self.hmac_key   = self.entropy_pool[32:64]

    def _build_dynamic_sbox(self, seed):
        """Generates a unique substitution table for this session."""
        sbox = list(range(256))
        for i in range(255, 0, -1):
            idx = int.from_bytes(hashlib.sha256(seed + bytes([i])).digest()[:2], 'big') % (i + 1)
            sbox[i], sbox[idx] = sbox[idx], sbox[i]
        return sbox

    @staticmethod
    def gf256_mult(a, b):
        """Galois Field multiplication (0x1B)."""
        res = 0
        for _ in range(8):
            if b & 1: res ^= a
            carry = a & 0x80
            a = (a << 1) & 0xFF
            if carry: a ^= 0x1B
            b >>= 1
        return res

    def mds_diffusion(self, block):
        """MDS Matrix [2, 3, 1, 1] diffusion layer."""
        b = list(block)
        return [
            self.gf256_mult(2, b[0]) ^ self.gf256_mult(3, b[1]) ^ b[2] ^ b[3],
            b[0] ^ self.gf256_mult(2, b[1]) ^ self.gf256_mult(3, b[2]) ^ b[3],
            b[0] ^ b[1] ^ self.gf256_mult(2, b[2]) ^ self.gf256_mult(3, b[3]),
            self.gf256_mult(3, b[0]) ^ b[1] ^ b[2] ^ self.gf256_mult(2, b[3])
        ]

    def process_block(self, plaintext_4byte):
        substituted = [self.sbox[b] for b in plaintext_4byte]
        return bytes(self.mds_diffusion(substituted))
